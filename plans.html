1. ภาพรวม

เพิ่มระบบบริหารจัดการข้อมูลในโปรเจกต์ Portfolio เพื่อให้แอดมินสามารถ เพิ่ม / แก้ไข / ลบโปรเจกต์ ได้เองผ่านหน้าเว็บ พร้อมระบบ เข้าสู่ระบบที่ปลอดภัย (Password hashing + JWT + Middleware) และเชื่อมข้อมูลจากหน้า /projects ให้ดึงข้อมูลจากฐานข้อมูลผ่าน API แทนข้อมูลแบบ hardcode

ฐานข้อมูล

PostgreSQL (database server ของตัวเอง) + Prisma ORM

ระบบ Auth

bcryptjs hash + JWT (httpOnly cookie) + Middleware ป้องกัน

อัปโหลดรูป

MinIO (server แยก) + Presigned URL upload ตรงจาก browser

3.2 Flow การอัปโหลดรูป (Local Disk)

1. Admin เลือกไฟล์ในฟอร์ม (drag & drop หรือ file input)
2. Client → POST /api/uploads  (มี JWT cookie)  Body: FormData { file }
3. Server ตรวจ type / ขนาด (png/jpg/webp/gif/svg, ≤ 5MB)
   → บันทึกลง data/uploads/{uuid}.{ext}
   → return { filename }
4. เก็บ filename ลง Project.image ในฐานข้อมูล
5. หน้าเว็บแสดงภาพผ่าน /api/uploads/{filename} (GET สาธารณะ, cache ยาว)

การลบไฟล์ (ตอนลบโปรเจกต์ หรือเปลี่ยนรูป):
- ลบโปรเจกต์ → DELETE /api/admin/projects/{id}  → server ลบไฟล์ให้ด้วย
- เปลี่ยนรูปใหม่ / เอาออก → server ลบไฟล์เก่าให้อัตโนมัติ
ข้อดี: ไม่ต้องตั้งค่า CORS / policy / external URL — ไฟล์อยู่บน server เดียว มี backup พร้อม disk เดียวกับแอป ใช้ data/ และใส่ใน .gitignore

5. Database Schema (Prisma)

// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  admin
}

enum ProjectStatus {
  completed
  in-progress
}

model User {
  id           String   @id @default(cuid())
  username     String   @unique
  passwordHash String
  role         Role     @default(admin)
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt
}

model Project {
  id          String        @id @default(cuid())
  name        String
  description String
  image       String?       // filename ใน data/uploads (เช่น abc123.webp)
  tags        String[]      @default([])
  categories  String[]      @default([])
  year        String
  status      ProjectStatus @default(completed)
  github      String?
  demo        String?
  createdAt   DateTime      @default(now())
  updatedAt   DateTime      @updatedAt

  @@index([status])
  @@index([categories])
}

6. API Routes

Method Path Access หน้าที่
GET /api/projects Public รายการโปรเจกต์ + ?category= ?search=
GET /api/projects/[id] Public ดึงโปรเจกต์เดี่ยว

POST /api/upload/presign 🔒 Admin สร้าง Presigned PUT URL + objectKey
DELETE /api/upload/[key] 🔒 Admin ลบ object ใน MinIO

7. Auth & Middleware

7.1 Password Hashing (bcryptjs)

// src/lib/auth.ts
import bcrypt from "bcryptjs";

const SALT_ROUNDS = 10;

export async function hashPassword(password: string) {
  return bcrypt.hash(password, SALT_ROUNDS);
}

export async function verifyPassword(password: string, hash: string) {
  return bcrypt.compare(password, hash);
}

7.2 JWT (jose)

// src/lib/jwt.ts
import { SignJWT, jwtVerify } from "jose";

const secret = new TextEncoder().encode(process.env.JWT_SECRET);

export async function signToken(payload: { userId: string; role: string }) {
  return new SignJWT(payload)
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime("7d")
    .sign(secret);
}

export async function verifyToken(token: string) {
  try {
    const { payload } = await jwtVerify(token, secret);
    return payload;
  } catch {
    return null;
  }
}

7.3 Middleware

// src/middleware.ts
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { verifyToken } from "@/lib/jwt";

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const token = request.cookies.get("token")?.value;
  const session = token ? await verifyToken(token) : null;

  // หน้า admin: ไม่มี session → redirect ไปหน้า login
  if (pathname.startsWith("/dashboard")) {
    if (!session) {
      const url = new URL("/dashboard", request.url);
      return NextResponse.redirect(url);
    }
    return NextResponse.next();
  }

  // API admin/upload: ไม่มี session → 401
  if (pathname.startsWith("/api/admin") || pathname.startsWith("/api/upload")) {
    if (!session) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }
    return NextResponse.next();
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/dashboard/:path*", "/api/admin/:path*", "/api/upload/:path*"],
};

8.2 ตัวอย่างโค้ด src/lib/storage.ts

import { randomUUID } from "crypto";
import { mkdir, writeFile, unlink, readFile } from "fs/promises";
import path from "path";

const UPLOADS_DIR = path.join(process.cwd(), "data", "uploads");

export const ALLOWED_IMAGE_TYPES: Record<string, string> = {
  "image/png": "png", "image/jpeg": "jpg",
  "image/webp": "webp", "image/gif": "gif",
  "image/svg+xml": "svg",
};

export const MAX_IMAGE_SIZE = 5 * 1024 * 1024;

export function isValidImageFilename(filename: string): boolean {
  return (
    filename.length > 0 && filename.length <= 255 &&
    path.basename(filename) === filename &&          // ไม่มี path traversal
    !filename.includes("..") &&
    /^[A-Za-z0-9][A-Za-z0-9._-]*$/.test(filename)
  );
}

export async function saveImage(buffer: Buffer, contentType: string) {
  const ext = ALLOWED_IMAGE_TYPES[contentType];
  if (!ext) throw new Error("Unsupported content type");
  const filename = `${randomUUID()}.${ext}`;
  await mkdir(UPLOADS_DIR, { recursive: true });
  await writeFile(path.join(UPLOADS_DIR, filename), buffer);
  return filename;
}

export async function deleteImage(filename: string) {
  if (!isValidImageFilename(filename)) return;
  try { await unlink(path.join(UPLOADS_DIR, filename)); } catch { /* ENOENT = โอเค */ }
}

10. Security Checklist

  • รหัสผ่านเก็บเป็น bcrypt hash เท่านั้น — ไม่เก็บ plain text
  • JWT เก็บใน httpOnly cookie (JavaScript อ่านไม่ได้) + sameSite: lax + secure ใน production + maxAge 7 วัน
  • Middleware ป้องกัน /dashboard/*, /api/admin/*, /api/uploads POST/DELETE — ไม่มี token → redirect/401 (GET รูปปล่อยสาธารณะ)
  • เช็ค role === "admin" ใน helper getAuthUser() สำหรับ API ที่ sensitive
  • ลบ password hardcode ใน src/app/dashboard/page.tsx ออก
  • เซ็ต env ทั้งหมดลง .env (มีใน .gitignore แล้ว) — ไม่ hardcode secret ในโค้ด
  • Validation ฝั่ง server ทุก endpoint: ชื่อ/คำอธิบายไม่ว่าง, status ตรง enum, ขนาดไฟล์ ≤ 5MB, allow-list content type (png/jpg/webp/gif/svg)
  • ตั้งชื่อไฟล์ใหม่ด้วย crypto.randomUUID() + ext เดิม — ป้องกัน path traversal และชื่อซ้ำ
  • Middleware matcher แคบที่สุดเท่าที่จำเป็น (เฉพาะ path ที่ต้องการ)
